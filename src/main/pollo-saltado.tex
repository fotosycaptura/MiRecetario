\begin{recipe}[source={Lizz},
	portion={2 porciones},
	preparationtime={\unit[30]{Minutos}}
	]{Pollo saltado}
	\includegraphics[width=0.25\textwidth]{pollo-saltado.png}
	\introduction{
		A falta de lomo, siempre se puede usar pollo, como lo verás en esta receta...
	}
	\ingredients{
		1 & pechuga de pollo deshuesada \\
		1 & cebolla morada \\
                1 & ají escabeche \\
                 & Salsa de soja \\
		 & Aceite de oliva \\
		 & Sal a gusto \\
                 & Salsa de ostión \\
                 & Pimienta a gusto \\
                 & Caldo de pollo \\
                 & Cilantro picado \\
	}
	\preparation{
		\begin{enumerate}
			\item Se corta los extremos de la cebolla, se retira la cáscara, la primera capa,  luego al centro y se extraen los corazones, luego se cortan los gajos de la cebolla en trozos desde dentro hacia fuera.
			\item El tomate se pela y se corta en cuatro, luego se extraen las semillas, esto para evitar que quede una sopa, además la salsa de esta manera se adhiere mejor al tomate pelado, quedando más jugoso.
                        \item Luego se toma un ají amarillo o escabeche, se cortan los extremos, y se le extraen las semillas. Posteriormente se corta en tiras finas o juliana, y luego a la mitad de ellas.
                        \item El pollo se corta en trozos de bocado, se sazona con sal, pimienta y se le echa un poco de aceite y se mezcla en un bol, para darle color, se le puede echar una cucharada de salsa de soja. Se puede dejar en el refrigerador por media hora o se puede usar de inmediato.
                        \item En una sartén bien caliente, se aplica un chorro de aceite, se agrega el pollo y se deja a fuego fuerte con la finalidad de sellar o dorar.
                        \item Luego se agrega la cebolla en gajos, ají amarillo en tiras y se sigue salteando a fuego fuerte. Recordar que la cebolla no se debe marchitar. Un par de vueltas y ya está. Agregar una cucharada de salsa de soja, un poco de vinagre, y si se desea, salsa de ostión.
                        \item Agregar el tomate en gajos. Para terminar, se integra todo, se agrega un poco de caldo de pollo. Se le puede agregar un poco de cilantro al gusto.
                        \item Llegados a este punto, Se puede servir.
		\end{enumerate}
	}
	\hint{
		Puedes acompañar con papas fritas y arroz blanco. Puedes usar una crema con ají amarillo para darle más sabor al arroz.
	}
\end{recipe}
